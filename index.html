// دالة تسجيل الدخول وتحديد الصلاحيات async function handleLogin(email, password) { try { // 1. المصادقة عبر Firebase Auth const userCredential = await firebase.auth().signInWithEmailAndPassword(email, password); const user = userCredential.user; // 2. جلب بيانات الصلاحية والمدرسة من Firestore const userDoc = await firebase.firestore().collection('users').doc(user.uid).get(); if (userDoc.exists) { const userData = userDoc.data(); // تخزين بيانات الجلسة محلياً localStorage.setItem('current_school', userData.schoolId); localStorage.setItem('user_role', userData.role); // 3. توجيه المستخدم حسب دوره if (userData.role === 'admin') { alert('مرحباً بك، مدير المدرسة (صلاحية كاملة لكافة المجالات)'); switchView('overview'); } else { // إذا كان مسؤول مجال (مثلاً domain1 للمجال الأول) const targetDomain = userData.role.replace('domain', ''); alert(`مرحباً بك، تم توجيهك لصلاحيات المجال رقم ${targetDomain}`); switchView(targetDomain); } } else { alert('حساب المستخدم غير مسجل في قاعدة بيانات المنظومة.'); } } catch (error) { console.error("خطأ في تسجيل الدخول:", error.message); alert("فشل تسجيل الدخول: " + error.message); } }
